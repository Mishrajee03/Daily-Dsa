class Solution {
    public int lengthOfLongestSubstring(String s) {
     int maxlen=0;
     int low=0;
     HashMap<Character,Integer> map=new HashMap<>();

     for(int i=0;i<s.length();i++){
        char ch=s.charAt(i);

        if(map.containsKey(ch)){
            low=Math.max(low,map.get(ch)+1);
        }

        map.put(ch,i);
        maxlen=Math.max(maxlen,i-low+1);
     }

     return maxlen;
    }
}
