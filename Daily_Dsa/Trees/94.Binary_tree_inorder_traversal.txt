/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public List<Integer> inorderTraversal(TreeNode root) {
        List<Integer> ans=new ArrayList<>();
        help(ans,root);
        return ans;

    }
    public List<Integer> help(List<Integer> res,TreeNode root){
        if(root!=null){
            help(res,root.left);
            res.add(root.val);
            help(res,root.right);
        }
        return res;
    }

}
